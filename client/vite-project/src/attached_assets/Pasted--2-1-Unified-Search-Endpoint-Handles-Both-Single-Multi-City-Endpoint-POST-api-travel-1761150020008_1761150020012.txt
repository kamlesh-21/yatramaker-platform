## 2.1 Unified Search Endpoint (Handles Both Single & Multi-City)

### Endpoint: `POST /api/travelData/recommendations`

### Request Payload

```javascript
{
  // ==========================================
  // COMMON FIELDS (Existing - Keep Same)
  // ==========================================
  "location": "Patna",
  "budget": 50000,
  "tripDuration": 7,
  "travelers": {
    "adults": 2,
    "children": 1,
    "infants": 0
  },
  "preferences": ["spiritual", "heritage", "cultural"],
  "accommodationPreference": "mid-range", // budget | mid-range | luxury
  
  // ==========================================
  // NEW FIELD: Trip Type Selector
  // ==========================================
  "tripType": "multi-city", // "single-destination" | "multi-city"
  
  // ==========================================
  // MULTI-CITY SPECIFIC (Only if tripType = "multi-city")
  // ==========================================
  "multiCityOptions": {
    "tripStructure": "loop", // linear | loop | hub-spoke
    
    "mustVisitDestinations": [ // Optional
      {
        "destination_id": 45,
        "name": "Bodh Gaya",
        "priority": "must" // must | prefer | optional
      },
      {
        "destination_id": 78,
        "name": "Nalanda",
        "priority": "prefer"
      }
    ],
    
    "pacingPreference": "moderate", // relaxed | moderate | intensive
    
    "luggageStyle": "minimize-changes", // minimize-changes | flexible
    
    "experiencePriorities": { // 0-10 scale, optional
      "historical": 8,
      "spiritual": 9,
      "adventure": 2,
      "nature": 5,
      "cultural": 7,
      "photography": 6,
      "shopping": 3,
      "food": 7,
      "relaxation": 4,
      "nightlife": 1
    },
    
    "optimizationGoal": "balanced" // cheapest | fastest | comfortable | experiential | balanced
  }
}
```
